#include<iostream>
using namespace std;

int sum(int n){
    int s=0;
    while(n>0){
        s+=n%10;
        n/=10;
    }
    return s;
}

int main(){
    int x,y,cnt=0;
    cin>>x>>y;
    for(int i=x;i<=y;i++){
        if(sum(i)%5==0)cnt++;
    }
    cout<<cnt<<endl;
    return 0;
}
