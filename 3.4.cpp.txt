#include <iostream>
using namespace std;

int main()
{
    int T;
    cin>>T;
    while(T--)
    {
    int n;
    cin>>n;
    int m=n/2;
    for(int i=0;i<=m;++i)
    {
     for(int j=0;j<m-i;++j)   
         cout<<" ";
        for(int j=0;j<2*i+1;++j)
            cout<<"*";
        cout<<endl;
    }
       for (int i=m-1;i>=0;--i)
        {
            for (int j=0;j<m-i;++j)
                cout << " ";
            for (int j=0; j<2*i +1;++j)
                cout<<"*";
            cout<<endl;
        }
    }
    return 0;
        }
