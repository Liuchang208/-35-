#include<iostream>
#include<string>
#include<algorithm>
using namespace std;
typedef unsigned long long ull;

bool isprime(ull n){
    if(n<=1)return false;
    if(n==2)return true;
    if(n%2==0)return false;
    for(ull i=3;i*i<=n;i+=2){
        if(n%i==0)return false;
    }
    return true;
}

int main(){
    string s;
    cin>>s;
    string tmp=s.substr(0,s.length()-1);
    reverse(tmp.begin(),tmp.end());
    string resstr=s+tmp;
    ull num=0;
    for(char c:resstr){
        num=num*10+(c-'0');
    }
    if(isprime(num))cout<<"prime\n";
    else cout<<"noprime\n";
    return 0;
}
