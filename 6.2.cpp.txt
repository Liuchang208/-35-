#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;
int main()
{
ios::sync_with_stdio(false);
cin.tie(0);
int n,K;
cin>>n>>K;
vector<long long> A(n),B(n),diff(n);
long long sumB=0;
for(int i=0;i<n;i++) cin>>A[i];
for(int i=0;i<n;i++)
{
cin>>B[i];
sumB+=B[i];
diff[i]=A[i]-B[i];
}
sort(diff.rbegin(),diff.rend());
for(int i=0;i<K;i++)
{
sumB+=diff[i];
}
cout<<sumB<<endl;
return 0;
}
