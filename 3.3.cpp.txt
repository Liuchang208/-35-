#include <iostream>
using namespace std;

int main()
{
    char c;
    cin>>c;
    int n=c-'A'+1;
    for(int i=0;i<n;i++)
    {
        for(int j=0;j < n - i - 1;j++)
            printf(" ");
        for(int j=0;j<=i;j++)
            printf("%c",(char)('A'+j));
        for(int j=i-1;j>=0;j--)
            printf("%c",(char)('A'+j));
        printf("\n");
          }
    return 0;
}
