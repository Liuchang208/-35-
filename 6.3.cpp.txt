#include<iostream>
#include<algorithm>
using namespace std;

int main()
{
int n;
cin>>n;
 long long a[105];
for(int i=0;i<n;i++)
    {
cin>>a[i];
}
sort(a,a+n);
bool ok=false;
long long x,y,z;
for(int i=0;i<n;i++)
{
for(int j=i+1;j<n;j++)
{
for(int k=j+1;k<n;k++)
 {
if(a[i]+a[j]>a[k])
 {
x=a[i];y=a[j];z=a[k];
ok=true;
goto endloop;
}
}
}
}
endloop:
    if(ok)
 {
        cout<<x<<" "<<y<<" "<<z<<endl;
    }
    else
    {
        cout<<"No solution"<<endl;
    }
    return 0;
}
