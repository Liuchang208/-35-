#include<bits/stdc++.h>
#include<string>
#include<iterator>

using namespace std;
int main(){
    string s((istreambuf_iterator<char>(cin)),istreambuf_iterator<char>());
    string res;
    enum state {nor,string,com};
    state state=nor;
    int n=s.size();
    for(int i=0;i<n;i++){
        char c=s[i];
        if(state==nor){
            if(c=='\\'&&i+1<n&&s[i+1]=='"')
            {
                res+="\\\"";
                i++;
            }
            else if(c=='"')
            {
                res+='"';
                state=string;
            }
            else if(c=='/'&&i+1<n&&s[i+1]=='*')
            {
                size_t pos=s.find("*/",i+2);
                if(pos!=string::npos)
                {
                    i=pos+1;
                }
                else{
                    res+="/*";
                    i++;
                }
            }else{
                res+=c;
            }
        }
        else if(state==string){
            if(c=='\\'&&i+1<n&&s[i+1]=='"')
            {
                res+="\\\"";
                i++;
            }
            else if(c=='"'
                   ){
                res+='"';
                state=nor;
            }
            else{
                res+=c;
            }
        }
    }
    cout << res;
}
