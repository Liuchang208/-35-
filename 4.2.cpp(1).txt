#include<iostream>
using namespace std;

int countDigit(int num,int x){
    int cnt=0;
    while(num>0){
        if(num%10==x)cnt++;
        num/=10;
    }
    return cnt;
}

int main(){
    int n,x,ans=0;
    cin>>n>>x;
    for(int i=1;i<=n;i++){
        ans+=countDigit(i,x);
    }
    cout<<ans<<endl;
    return 0;
}
