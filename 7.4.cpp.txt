#include<iostream>
#include<string>
using namespace std;
int main()
{
    int T;
    cin>>T;
    while(T--)
    {
        string s;
        cin>>s;
        string ans;
        int i=0;
        int len=s.size();
        while(i<len)
        {
            if(s[i]=='0'){ans+='0';i++;}
            else if(s[i]=='2'){ans+='2';i++;}
            else if(s[i]=='4'){ans+='4';i++;}
            else if(s[i]=='6'){ans+='6';i++;}
            else if(s[i]=='9'){ans+='9';i++;}
            else if(i+3<len&&s.substr(i,4)=="zero"){ans+='0';i+=4;}
            else if(i+2<len&&s.substr(i,3)=="two"){ans+='2';i+=3;}
            else if(i+3<len&&s.substr(i,4)=="four"){ans+='4';i+=4;}
            else if(i+2<len&&s.substr(i,3)=="six"){ans+='6';i+=3;}
            else if(i+2<len&&s.substr(i,4)=="nine"){ans+='9';i+=4;}
            else{i++;}
        }
        cout<<ans<<endl;
    }
    return 0;
}
