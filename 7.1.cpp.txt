#include <iostream>
#include <string>
using namespace std;
int main()
{
    int n;
    long long t;
    cin>>n>>t;
    string s;
    cin>>s;
    int dot=-1;
    for(int i=0;i<s.size();i++)
    {
        if(s[i]=='.')
        {
            dot=i;
            break;
        }
    }
    if(dot==-1)
    {
        cout<<s<<endl;
        return 0;
    }

    while(t>0)
    {
        int find=-1;
        for(int i=dot+1;i<s.size();i++)
        {
            if(s[i]>='5')
            {
                find=i;
                break;
            }
        }
        if(find==-1) break;

        t--;
        s.erase(find);
        int p=find-1;
        int carry=1;
        while(carry && p>=0)
        {
            if(s[p]=='.')
            {
                p--;
                continue;
            }
            int d=s[p]-'0';
            d+=carry;
            if(d==10)
            {
                s[p]='0';
                carry=1;
                p--;
            }
            else
            {
                s[p]=d+'0';
                carry=0;
            }
        }
        if(carry)
        {
            s.insert(0,"1");
            dot++;
        }
    }

    while(s.size()>dot+1 && s.back()=='0')
    {
        s.pop_back();
    }
    if(s.back()=='.')
    {
        s.pop_back();
    }
    cout<<s<<endl;
    return 0;
}
