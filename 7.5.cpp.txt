#include<bits/stdc++.h>
using namespace std;
int sum[10];
int a=0,b=0;
int ans[100]={0};
int cnt=0;
int k=-1;
int biaoji=0;
int main()
{
    for(int i=0;i<10;i++)
    {
        cin>>sum[i];
        if(sum[i]==0)continue;
        if(sum[i]%2==0&&i!=0)b++;
        else if(sum[i]%2!=0){a++;k=i;}
    }
    if(a>1){cout<<"-1";return 0;}
    if(sum[0]>1&&b==0&&k<=0){cout<<"-1";return 0;}
    if(sum[0]>1&&b==0&&k>0&&sum[k]==1){cout<<"-1";return 0;}
    for(int i=1;i<10;i++)
    {
        int t=sum[i]/2;
        for(int j=0;j<t;j++)
        {
            ans[cnt++]=i;
            if(biaoji!=1){cnt+=sum[0]/2;biaoji=1;}
        }
    }
    for(int i=0;i<cnt;i++)cout<<ans[i];
    if(k>=0)cout<<k;
    for(int i=cnt-1;i>=0;i--)cout<<ans[i];
}
