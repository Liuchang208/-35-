#include <iostream>
using namespace std;

int main()
{
    int L,R;
    cin>>L>>R;
    int cnt=0;
    for(int i=L;i<=R;i++)
{
        int x=i;
        while(x>0)
{
        int digit=x%10;
            if(digit==2)
            {
               cnt++;
            }    
            x=x/10;
        }
    }
    cout<<cnt<<endl;
    return 0;
}
