#include<iostream>
using namespace std;

int main(){
    int L,M;
    cin>>L>>M;
    bool tree[10005]={0};
    for(int i=0;i<=L;i++)tree[i]=true;
    for(int i=0;i<M;i++){
        int l,r;
        cin>>l>>r;
        for(int j=l;j<=r;j++)tree[j]=false;
    }
    int cnt=0;
    for(int i=0;i<=L;i++)if(tree[i])cnt++;
    cout<<cnt<<endl;
    return 0;
}
