#include<iostream>
#include<string>
#include<cstring>
using namespace std;
int jump[10][10];
bool vis[10];
void init()
{
memset(jump,0,sizeof(jump));
jump[1][3]=2;jump[3][1]=2;
jump[1][7]=4;jump[7][1]=4;
jump[1][9]=5;jump[9][1]=5;
jump[2][8]=5;jump[8][2]=5;
jump[3][7]=5;jump[7][3]=5;
jump[3][9]=6;jump[9][3]=6;
jump[4][6]=5;jump[6][4]=5;
jump[7][9]=8;jump[9][7]=8;
}
int main()
{
init();
string s;
while(cin>>s)
{
memset(vis,0,sizeof(vis));
bool ok=true;
for(int i=0;i<s.size();i++)
{
int cur=s[i]-'0';
if(vis[cur])
{
ok=false;
break;
}
vis[cur]=true;
if(i==0)continue;
int pre=s[i-1]-'0';
int mid=jump[pre][cur];
if(mid!=0&&!vis[mid])
{
ok=false;
break;
}
}
if(ok)cout<<"YES\n";
else cout<<"NO\n";
}
return 0;
}
